// Level design (0=empty, 1=grass, 2=dirt, 3=stone, 4=brick, 5=question, 6=pipeTop, 7=pipeBody, 9=flagPole, 10=flagTop)

const TILE_SIZE = 32;

const LEVEL_WIDTH = 200;

const LEVEL_HEIGHT = 15;

function generateLevel() {

    const level = [];

    for (let y = 0; y < LEVEL_HEIGHT; y++) {

        level[y] = [];

        for (let x = 0; x < LEVEL_WIDTH; x++) {

            level[y][x] = BLOCK.EMPTY;

        }

    }

    // Ground layer

    for (let x = 0; x < LEVEL_WIDTH; x++) {

        level[LEVEL_HEIGHT-1][x] = BLOCK.GRASS;

        level[LEVEL_HEIGHT-2][x] = BLOCK.DIRT;

        level[LEVEL_HEIGHT-3][x] = BLOCK.DIRT;

    }

    // Some gaps

    for (let x = 45; x < 50; x++) {

        level[LEVEL_HEIGHT-1][x] = BLOCK.EMPTY;

        level[LEVEL_HEIGHT-2][x] = BLOCK.EMPTY;

        level[LEVEL_HEIGHT-3][x] = BLOCK.EMPTY;

    }

    for (let x = 95; x < 100; x++) {

        level[LEVEL_HEIGHT-1][x] = BLOCK.EMPTY;

        level[LEVEL_HEIGHT-2][x] = BLOCK.EMPTY;

        level[LEVEL_HEIGHT-3][x] = BLOCK.EMPTY;

    }

    for (let x = 145; x < 150; x++) {

        level[LEVEL_HEIGHT-1][x] = BLOCK.EMPTY;

        level[LEVEL_HEIGHT-2][x] = BLOCK.EMPTY;

        level[LEVEL_HEIGHT-3][x] = BLOCK.EMPTY;

    }

    // Brick platforms

    function addPlatform(x, y, width) {

        for (let i = 0; i < width; i++) {

            if (y < LEVEL_HEIGHT) level[y][x+i] = BLOCK.BRICK;

        }

    }

    function addQuestion(x, y) {

        if (y < LEVEL_HEIGHT) level[y][x] = BLOCK.QUESTION;

    }

    // Section 1 platforms

    addPlatform(10, 10, 5);

    addQuestion(12, 10);

    addPlatform(18, 8, 3);

    addQuestion(19, 8);

    addPlatform(25, 10, 4);

    // Section 2

    addPlatform(55, 9, 6);

    addQuestion(57, 9);

    addQuestion(58, 9);

    addPlatform(65, 7, 4);

    addPlatform(72, 10, 5);

    addQuestion(74, 10);

    // Section 3

    addPlatform(105, 9, 5);

    addQuestion(107, 9);

    addPlatform(112, 7, 4);

    addPlatform(120, 10, 6);

    addQuestion(122, 10);

    addQuestion(123, 10);

    // Section 4 - Boss area

    addPlatform(155, 10, 8);

    addPlatform(165, 8, 10);

    addPlatform(175, 10, 8);

    // Pipes

    function addPipe(x, y, height) {

        level[y][x] = BLOCK.PIPE_TOP;

        level[y][x+1] = BLOCK.PIPE_TOP;

        for (let h = 1; h < height; h++) {

            level[y+h][x] = BLOCK.PIPE_BODY;

            level[y+h][x+1] = BLOCK.PIPE_BODY;

        }

    }

    addPipe(30, 11, 2);

    addPipe(60, 10, 3);

    addPipe(80, 11, 2);

    addPipe(110, 11, 2);

    addPipe(130, 10, 3);

    // Flag at the end

    level[11][LEVEL_WIDTH-5] = BLOCK.FLAG_POLE;

    level[10][LEVEL_WIDTH-5] = BLOCK.FLAG_POLE;

    level[9][LEVEL_WIDTH-5] = BLOCK.FLAG_POLE;

    level[8][LEVEL_WIDTH-5] = BLOCK.FLAG_POLE;

    level[7][LEVEL_WIDTH-5] = BLOCK.FLAG_POLE;

    level[6][LEVEL_WIDTH-5] = BLOCK.FLAG_POLE;

    level[5][LEVEL_WIDTH-5] = BLOCK.FLAG_POLE;

    level[4][LEVEL_WIDTH-5] = BLOCK.FLAG_POLE;

    level[3][LEVEL_WIDTH-5] = BLOCK.FLAG_TOP;

    return level;

}

// Enemies

class Enemy {

    constructor(x, y, type) {

        this.x = x;

        this.y = y;

        this.width = 26;

        this.height = 26;

        this.vx = type === 'boss' ? 1.5 : 1;

        this.vy = 0;

        this.type = type; // 'goomba', 'turtle', 'fly', 'boss'

        this.dead = false;

        this.direction = -1;

        this.animFrame = 0;

        this.hp = type === 'boss' ? 10 : 1;

        this.maxHp = type === 'boss' ? 10 : 1;

        this.bossPhase = 0;

        this.bossTimer = 0;

    }

    update() {

        if (this.dead) return;

        this.animFrame++;

        if (this.type === 'boss') {

            this.bossTimer++;

            // Boss movement pattern

            if (this.bossTimer % 180 < 90) {

                this.vx = 1.5 * this.direction;

            } else if (this.bossTimer % 180 < 120) {

                this.vx = 0;

                // Jump attack

                if (this.bossTimer % 180 === 90 && this.grounded) {

                    this.vy = -14;

                }

            } else {

                this.vx = 2.5 * this.direction;

            }

            // Change direction at edges

            if (this.x < 160 * TILE_SIZE || this.x > 190 * TILE_SIZE) {

                this.direction *= -1;

            }

        } else if (this.type === 'fly') {

            this.vy = Math.sin(this.animFrame * 0.05) * 2;

            this.x += this.vx * this.direction;

        } else {

            this.x += this.vx * this.direction;

        }

        // Gravity for non-flying enemies

        if (this.type !== 'fly') {

            this.vy += 0.5;

            this.y += this.vy;

        } else {

            this.y += this.vy;

        }

        // Platform collision

        this.grounded = false;

        const startX = Math.floor(this.x / TILE_SIZE);

        const endX = Math.floor((this.x + this.width) / TILE_SIZE);

        const startY = Math.floor(this.y / TILE_SIZE);

        const endY = Math.floor((this.y + this.height) / TILE_SIZE);

        for (let ty = startY; ty <= endY; ty++) {

            for (let tx = startX; tx <= endX; tx++) {

                if (ty >= 0 && ty < LEVEL_HEIGHT && tx >= 0 && tx < LEVEL_WIDTH) {

                    const block = level[ty][tx];

                    if (block !== BLOCK.EMPTY && block !== BLOCK.COIN &&

                        block !== BLOCK.FLAG_POLE && block !== BLOCK.FLAG_TOP) {

                        // Simple collision resolution

                        if (this.vy > 0 && this.y + this.height > ty * TILE_SIZE &&

                            this.y < ty * TILE_SIZE) {

                            this.y = ty * TILE_SIZE - this.height;

                            this.vy = 0;

                            this.grounded = true;

                        }

                    }

                }

            }

        }

        // Turn around at edges or walls

        if (this.type !== 'boss') {

            const frontX = this.direction > 0 ?

                Math.floor((this.x + this.width + 2) / TILE_SIZE) :

                Math.floor((this.x - 2) / TILE_SIZE);

            const groundY = Math.floor((this.y + this.height + 2) / TILE_SIZE);

            if (frontX < 0 || frontX >= LEVEL_WIDTH ||

                (groundY < LEVEL_HEIGHT && level[groundY][frontX] === BLOCK.EMPTY)) {

                this.direction *= -1;

            }

        }

    }

    draw() {

        if (this.dead) return;

        const screenX = this.x - camera.x;

        const screenY = this.y - camera.y;

        if (screenX < -50 || screenX > canvas.width + 50) return;

        ctx.save();

        ctx.translate(screenX, screenY);

        if (this.type === 'goomba') {

            // Goomba - brown mushroom-like

            ctx.fillStyle = '#8B4513';

            ctx.fillRect(2, 8, 22, 18);

            ctx.fillStyle = '#A0522D';

            ctx.fillRect(4, 10, 18, 14);

            // Feet

            ctx.fillStyle = '#000';

// Particles

let particles = [];

class Particle {

    constructor(x, y, color, type) {

        this.x = x;

        this.y = y;

        this.vx = (Math.random() - 0.5) * 6;

        this.vy = (Math.random() - 1) * 6;

        this.life = 30;

        this.color = color;

        this.type = type;

        this.size = Math.random() * 4 + 2;

    }

    update() {

        this.x += this.vx;

        this.y += this.vy;

        this.vy += 0.3;

        this.life--;

    }

    draw() {

        const screenX = this.x - camera.x;

        const screenY = this.y - camera.y;

        ctx.fillStyle = this.color;

        ctx.fillRect(screenX, screenY, this.size, this.size);

    }

}

    gradient.addColorStop(0, '#87CEEB');

    gradient.addColorStop(0.5, '#B0E0E6');

    gradient.addColorStop(1, '#E0F6FF');

    ctx.fillStyle = gradient;

    ctx.fillRect(0, 0, canvas.width, canvas.height);

    // Clouds

    ctx.fillStyle = 'rgba(255, 255, 255, 0.8)';

    const cloudOffset = camera.x * 0.3;

    for (let i = 0; i < 8; i++) {

        const cx = ((i * 250) - cloudOffset) % (canvas.width + 300);

        const x = cx < -150 ? cx + canvas.width + 300 : cx;

        ctx.beginPath();

        ctx.arc(x + 25, y - 10, 35, 0, Math.PI * 2);

        ctx.arc(x + 50, y, 30, 0, Math.PI * 2);

        ctx.arc(x + 25, y + 10, 25, 0, Math.PI * 2);

        ctx.fill();

    }

    ctx.fillStyle = '#2d8a3e';

    for (let i = 0; i < 10; i++) {

        const hx = ((i * 400) - camera.x * 0.5) % (canvas.width + 400);

        ctx.beginPath();

        ctx.moveTo(x, canvas.height);

        ctx.lineTo(x + 100, canvas.height - 120);

// Collision detection

function checkCollision(rect1, rect2) {

    return rect1.x < rect2.x + rect2.width &&

           rect1.x + rect1.width > rect2.x &&

           rect1.y < rect2.y + rect2.height &&

           rect1.y + rect1.height > rect2.y;

}

// Player physics

function updatePlayer() {

    if (player.dead) {

        player.vy += 0.5;

        player.y += player.vy;

        if (player.y > canvas.height + 100) {

            lives--;

            if (lives <= 0) {

                gameOver();

            } else {

                respawnPlayer();

            }

        }

        return;

    }

    if (player.won) {

        player.vx *= 0.9;

        player.x += player.vx;

        // Slide down flag

        if (player.y < (LEVEL_HEIGHT-4) * TILE_SIZE) {

            player.y += 2;

        }

        return;

    }

    // Input

    if (keys['ArrowLeft']) {

        player.vx = Math.max(player.vx - 0.5, -player.speed);

        player.facing = -1;

    } else if (keys['ArrowRight']) {

        player.vx = Math.min(player.vx + 0.5, player.speed);

        player.facing = 1;

    } else {

        player.vx *= 0.8;

    }

    if (keys['Space'] && player.grounded) {

        player.vy = -player.jumpPower;

        player.grounded = false;

        // Jump particles

        for (let i = 0; i < 5; i++) {

            particles.push(new Particle(player.x + player.width/2, player.y + player.height, '#8B6914', 'dust'));

        }

    }

    // Gravity

    player.vy += player.gravity;

    // Apply velocity

    player.x += player.vx;

    player.y += player.vy;

    // Level boundaries

    if (player.x < 0) player.x = 0;

    if (player.x > LEVEL_WIDTH * TILE_SIZE - player.width) player.x = LEVEL_WIDTH * TILE_SIZE - player.width;

    // Platform collision

    player.grounded = false;

    const startX = Math.floor(player.x / TILE_SIZE);

    const endX = Math.floor((player.x + player.width) / TILE_SIZE);

    const startY = Math.floor(player.y / TILE_SIZE);

    const endY = Math.floor((player.y + player.height) / TILE_SIZE);

    for (let ty = startY; ty <= endY; ty++) {

        for (let tx = startX; tx <= endX; tx++) {

            if (ty >= 0 && ty < LEVEL_HEIGHT && tx >= 0 && tx < LEVEL_WIDTH) {

                const block = level[ty][tx];

                if (block !== BLOCK.EMPTY && block !== BLOCK.COIN &&

                    block !== BLOCK.FLAG_POLE && block !== BLOCK.FLAG_TOP) {

                    const blockRect = {

                        x: tx * TILE_SIZE,

                        y: ty * TILE_SIZE,

                        width: TILE_SIZE,

                        height: TILE_SIZE

                    };

                    if (checkCollision(player, blockRect)) {

                        // Resolve collision

                        const overlapX = (player.x + player.width/2) - (blockRect.x + blockRect.width/2);

                        const overlapY = (player.y + player.height/2) - (blockRect.y + blockRect.height/2);

                        const combinedHalfWidths = player.width/2 + blockRect.width/2;

                        const combinedHalfHeights = player.height/2 + blockRect.height/2;

                        const overlapXAbs = combinedHalfWidths - Math.abs(overlapX);

                        const overlapYAbs = combinedHalfHeights - Math.abs(overlapY);

                        if (overlapXAbs < overlapYAbs) {

                            // Horizontal collision

                            if (overlapX > 0) {

                                player.x = blockRect.x + blockRect.width;

                            } else {

                                player.x = blockRect.x - player.width;

                            }

                            player.vx = 0;

                        } else {

                            // Vertical collision

                            if (overlapY > 0) {

                                player.y = blockRect.y + blockRect.height;

                                player.vy = 0;

                                // Hit block from below

                                if (block === BLOCK.QUESTION) {

                                    level[ty][tx] = BLOCK.BRICK;

                                    score += 200;

                                    // Spawn coin

                                    for (let c of levelCoins) {

                                        if (!c.collected && Math.abs(c.x - tx * TILE_SIZE) < 20) {

                                            c.collected = true;

                                            coins++;

                                            score += 100;

                                            for (let i = 0; i < 8; i++) {

                                                particles.push(new Particle(tx * TILE_SIZE + TILE_SIZE/2, ty * TILE_SIZE, '#FFD700', 'coin'));

                                            }

                                            break;

                                        }

                                    }

                                }

                            } else {

                                player.y = blockRect.y - player.height;

                                player.vy = 0;

                                player.grounded = true;

                            }

                        }

                    }

                } else if (block === BLOCK.FLAG_POLE || block === BLOCK.FLAG_TOP) {

                    const flagRect = {

                        x: tx * TILE_SIZE,

                        y: ty * TILE_SIZE,

                        width: TILE_SIZE,

                        height: TILE_SIZE

                    };

                    if (checkCollision(player, flagRect)) {

                        winGame();

                    }

                }

            }

        }

    }

    // Fall off world

    if (player.y > LEVEL_HEIGHT * TILE_SIZE) {

        player.dead = true;

        player.vy = -5;

    }

    // Invincibility

    if (player.invincible > 0) player.invincible--;

    // Camera follow

    camera.x = player.x - canvas.width / 3;

    camera.x = Math.max(0, Math.min(camera.x, LEVEL_WIDTH * TILE_SIZE - canvas.width));

}

// Enemy interactions

function updateEnemies() {

    enemies.forEach(enemy => {

        enemy.update();

        if (enemy.dead) return;

        // Player collision

        if (player.invincible <= 0 && !player.dead && !player.won &&

            checkCollision(player, enemy)) {

            if (enemy.type === 'boss') {

                // Boss damage - must jump on head

                if (player.vy > 0 && player.y + player.height < enemy.y + enemy.height/2) {

                    enemy.hp--;

                    player.vy = -8;

                    score += 500;

                    for (let i = 0; i < 10; i++) {

                        particles.push(new Particle(enemy.x + enemy.width/2, enemy.y, '#ff0000', 'hit'));

                    }

                    if (enemy.hp <= 0) {

                        enemy.dead = true;

                        score += 5000;

                        for (let i = 0; i < 30; i++) {

                            particles.push(new Particle(enemy.x + enemy.width/2, enemy.y + enemy.height/2, '#FFD700', 'explosion'));

                        }

                    }

                } else {

                    player.dead = true;

                    player.vy = -8;

                    player.vx = enemy.direction * 3;

                }

            } else {

                // Regular enemy

                if (player.vy > 0 && player.y + player.height < enemy.y + enemy.height/2) {

                    // Stomp enemy

                    enemy.dead = true;

                    player.vy = -6;

                    score += 100;

                    for (let i = 0; i < 8; i++) {

                        particles.push(new Particle(enemy.x + enemy.width/2, enemy.y + enemy.height/2, '#8B4513', 'stomp'));

                    }

                } else {

                    // Take damage

                    player.dead = true;

                    player.vy = -8;

                    player.vx = enemy.direction * 3;

                }

            }

        }

    });

    // Remove dead enemies

    enemies = enemies.filter(e => !e.dead || e.animFrame < e.animFrame + 60);

}

// Coin collection

function updateCoins() {

    levelCoins.forEach(coin => {

        if (coin.collected) return;

        if (checkCollision(player, coin)) {

            coin.collected = true;

            coins++;

            score += 100;

            for (let i = 0; i < 6; i++) {

                particles.push(new Particle(coin.x + coin.width/2, coin.y + coin.height/2, '#FFD700', 'collect'));

            }

        }

    });

}

// Particles

function updateParticles() {

    particles.forEach(p => p.update());

    particles = particles.filter(p => p.life > 0);

}

function drawParticles() {

    particles.forEach(p => p.draw());

}

// UI update

function updateUI() {

    document.getElementById('score').textContent = `SCORE: ${score.toString().padStart(6, '0')}`;

    document.getElementById('coins').textContent = `COINS: ${coins.toString().padStart(2, '0')}`;

    document.getElementById('lives').textContent = `LIVES: ${lives}`;

}

// Game states

function startGame() {

    document.getElementById('startScreen').classList.add('hidden');

    document.getElementById('gameOverScreen').classList.add('hidden');

    document.getElementById('winScreen').classList.add('hidden');

    score = 0;

    coins = 0;

    lives = 3;

    player.dead = false;

    player.won = false;

    player.x = 100;

    player.y = 200;

    player.vx = 0;

    player.vy = 0;

    player.invincible = 0;

    level = generateLevel();

    generateCoins();

    spawnEnemies();

    particles = [];

    gameRunning = true;

    gameLoop();

}

function restartGame() {

    startGame();

}

function respawnPlayer() {

    player.x = Math.max(100, camera.x + 50);

    player.y = 200;

    player.vx = 0;

    player.vy = 0;

    player.dead = false;

    player.invincible = 120;

}

function gameOver() {

    gameRunning = false;

    document.getElementById('finalScore').textContent = `Score: ${score}`;

    document.getElementById('gameOverScreen').classList.remove('hidden');

}

function winGame() {

    player.won = true;

    score += 5000;

    gameRunning = false;

    setTimeout(() => {

        document.getElementById('winScore').textContent = `Score: ${score}`;

        document.getElementById('winScreen').classList.remove('hidden');

    }, 2000);

}

// Main game loop

function gameLoop() {

    if (!gameRunning) return;

    frameCount++;

    // Clear

    ctx.clearRect(0, 0, canvas.width, canvas.height);

    // Update

    updatePlayer();

    updateEnemies();

    updateCoins();

    updateParticles();

    // Draw

    drawBackground();

    drawLevel();

    drawCoins();

    enemies.forEach(e => e.draw());

    drawParticles();

    drawPlayer();

    updateUI();

    requestAnimationFrame(gameLoop);

}

ctx.fillRect(0, 0, canvas.width, canvas.height);
